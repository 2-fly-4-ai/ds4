#define BUFFER_CAPACITY 128
typedef struct RingBuffer { unsigned head; unsigned tail; int values[BUFFER_CAPACITY]; } RingBuffer;
static int ring_push(RingBuffer *buffer, int value) {
    unsigned next = (buffer->head + 1u) % BUFFER_CAPACITY;
    if (next == buffer->tail) return 0;
    buffer->values[buffer->head] = value;
    buffer->head = next;
    return 1;
}
static int dequeue(RingBuffer *buffer, int *value) {
    if (buffer->tail == buffer->head) return 0;
    *value = buffer->values[buffer->tail];
    buffer->tail = (buffer->tail + 1u) % BUFFER_CAPACITY;
    return 1;
}
